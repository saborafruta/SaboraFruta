async function cadastrarFornecedorRapido(state, form) { state.fornecedorSalvando=true; state.fornecedorErro=''; try { const response=await fetch('{% url "cadastros:fornecedor-ajax-create" %}', { method:'POST', body:new FormData(form), headers:{'X-Requested-With':'XMLHttpRequest'}, }); const data=await response.json(); if(!response.ok){ const messages=Object.values(data.errors||{}).flat(); throw new Error(messages.join(' ')||data.erro||'Não foi possível cadastrar o fornecedor.'); } window.dispatchEvent(new CustomEvent('finance-select-created', { detail:{field:'fornecedor',value:data.id,label:data.label,search:data.search}, })); form.reset(); state.fornecedorTipo='J'; state.fornecedorModal=false; } catch(error) { state.fornecedorErro=error.message||'Não foi possível cadastrar o fornecedor.'; } finally { state.fornecedorSalvando=false; } } function formatarCpfCnpjFornecedor(digitos, tipo) { if(tipo==='F'){ return digitos .replace(/(\d{3})(\d)/,'$1.$2') .replace(/(\d{3})(\d)/,'$1.$2') .replace(/(\d{3})(\d{1,2})$/,'$1-$2'); } return digitos .replace(/(\d{2})(\d)/,'$1.$2') .replace(/(\d{3})(\d)/,'$1.$2') .replace(/(\d{3})(\d)/,'$1/$2') .replace(/(\d{4})(\d{1,2})$/,'$1-$2'); } function mascararCpfCnpjFornecedor(input, tipo) { const limite=tipo==='F'?11:14; const digitos=(input.value||'').replace(/\D/g,''); input.value=digitos.length>limite?digitos:formatarCpfCnpjFornecedor(digitos,tipo); } function remascararCpfCnpjFornecedor(input, tipo) { if(!input)return; mascararCpfCnpjFornecedor(input,tipo); } function colarCpfCnpjFornecedor(event, state) { event.preventDefault(); const texto=event.clipboardData?.getData('text')||''; const digitos=texto.replace(/\D/g,''); if(digitos.length===11)state.fornecedorTipo='F'; else if(digitos.length===14)state.fornecedorTipo='J'; event.target.value=(digitos.length===11||digitos.length===14) ?formatarCpfCnpjFornecedor(digitos,state.fornecedorTipo) :digitos.slice(0,18); } async function consultarCepFornecedor(state, form) { const cep=(form.elements.cep.value||'').replace(/\D/g,''); if(!cep)return; if(cep.length!==8){ state.fornecedorErro='CEP deve ter 8 dígitos.'; return; } state.cepConsultando=true; state.fornecedorErro=''; try { const response=await fetch('{% url "cadastros:consultar-cep" %}?cep='+encodeURIComponent(cep)); const data=await response.json(); if(!response.ok)throw new Error(data.erro||'CEP não encontrado.'); ['cep','endereco','bairro','cidade','uf','codigo_municipio_ibge'].forEach(field=>{ if(form.elements[field]&&data[field])form.elements[field].value=data[field]; }); } catch(error) { state.fornecedorErro=error.message||'Não foi possível consultar o CEP.'; } finally { state.cepConsultando=false; } }
